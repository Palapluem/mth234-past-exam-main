\question \textbf{(8 คะแนน)} จงตอบคำถามต่อไปนี้
\begin{parts}
    \part \textbf{(2 คะแนน)} ให้ $A = \begin{bmatrix} 2 & -1 & 5 \\ 3 & 0 & -2 \\ 1 & 4 & -3 \end{bmatrix}$ เป็น\mbox{เมทริกซ์}มาตรฐาน (standard matrix) สำหรับ\mbox{การแปลงเชิงเส้น} (\mbox{Linear Transformation}) $T : \mathbb{R}^3 \to \mathbb{R}^3$ จงหาค่าของ $T\left(\begin{bmatrix} 1 \\ -2 \\ 3 \end{bmatrix}\right)$
    \vspace{\stretch{1}}
    
    \part \textbf{(2 คะแนน)} กำหนดให้\mbox{เมทริกซ์} $B = \begin{bmatrix} 1 & 0 & 0 & 0 \\ 0 & 3 & 0 & 0 \\ 0 & 0 & -2 & 1 \\ 0 & 0 & 0 & -2 \end{bmatrix}$ สามารถเขียนในรูปแบบของจอร์แดน \newline (Jordan form) เป็น $\begin{bmatrix} M_1 & 0 \\ 0 & M_2 \end{bmatrix}$ จงระบุ\mbox{เมทริกซ์} $M_1$ และ $M_2$ สำหรับ\mbox{เมทริกซ์}นี้
    \vspace{\stretch{1}}
    
    \newpage
    \part \textbf{(2 คะแนน)} จงหา\mbox{เวกเตอร์}เจาะจงทั่วไป ซึ่งมีลำดับขั้นเท่ากับ 2 (generalized eigenvector of rank 2) \mbox{ที่สัมพันธ์กับค่าเจาะจง} (eigenvalue) $\lambda = 3$ ของ\mbox{เมทริกซ์} $C = \begin{bmatrix} 4 & -1 \\ 1 & 2 \end{bmatrix}$
    \vspace{\stretch{1}}
    
    \part \textbf{(2 คะแนน)} กำหนด\mbox{เมทริกซ์} $D = \begin{bmatrix} 7 & -4 \\ 5 & -2 \end{bmatrix}$ จงหาสมการลักษณะเฉพาะ (characteristic equation) และค่าเจาะจง (eigenvalues) ทั้งหมดของ\mbox{เมทริกซ์} $D$
    \vspace{\stretch{1}}
\end{parts}

% Question 2
\question \textbf{(12 คะแนน)} กำหนดให้ $T : \mathbb{R}^4 \to \mathbb{R}^3$ นิยามโดย
\[ T\left(\begin{bmatrix} x_1 \\ x_2 \\ x_3 \\ x_4 \end{bmatrix}\right) = \begin{bmatrix} x_1 - x_2 + x_3 + x_4 \\ x_1 + 2x_2 - x_3 + x_4 \\ 2x_1 + x_2 + 2x_4 \end{bmatrix} \]
\begin{parts}
    \part \textbf{(3 คะแนน)} จงแสดงว่า $T$ เป็นการแปลงเชิงเส้น
    \vspace{\stretch{1}}
    \part \textbf{(3 คะแนน)} จงหาฐานหลัก (basis) ของ $Ker(T)$ และค่าของ $nullity(T)$
    \vspace{\stretch{1}}
    \newpage
    \part \textbf{(3 คะแนน)} จงหาฐานหลัก (basis) ของ $Range(T)$ และค่าของ $rank(T)$
    \vspace{\stretch{1}}
    \part \textbf{(3 คะแนน)} จงตรวจสอบว่า $T$ เป็นฟังก์ชันหนึ่งต่อหนึ่ง (one-to-one) \mbox{หรือไม่} และเป็นฟังก์ชันทั่วถึง (onto) \mbox{หรือไม่}
    \vspace{\stretch{1}}
\end{parts}

% Question 3
\question \textbf{(12 คะแนน)} จงพิจารณาว่า \mbox{เมทริกซ์} $A$ ที่กำหนดให้ต่อไปนี้เป็น\mbox{เมทริกซ์}ที่แปลงให้เป็น\mbox{เมทริกซ์}\mbox{ทแยงมุม}ได้\mbox{หรือไม่} (Determine if $A$ is diagonalizable.)
\[ A = \begin{bmatrix} 3 & -1 & 0 \\ -2 & 4 & 0 \\ 1 & 2 & 1 \end{bmatrix} \]
\begin{parts}
    \part \textbf{(2 คะแนน)} จงหาค่าเจาะจง (eigenvalues) ทั้งหมดของ $A$
    \vspace{\stretch{1}}
    \part \textbf{(4 คะแนน)} จงหาฐานหลักของปริภูมิเจาะจง (basis for eigenspace) สำหรับแต่ละค่าเจาะจง
    \vspace{\stretch{1}}
    \newpage
    \part \textbf{(4 คะแนน)} หาก $A$ เป็น\mbox{เมทริกซ์}ที่แปลงให้เป็น\mbox{เมทริกซ์}\mbox{ทแยงมุม}ได้ จงทำการหา\mbox{เมทริกซ์} $P$ และ\mbox{เมทริกซ์}\linebreak \mbox{ทแยงมุม} (diagonal matrix) $D$ ซึ่งทำให้ $A = PDP^{-1}$ (ถ้าแปลงไม่ได้ให้ระบุเหตุผล)
    \vspace{\stretch{1}}
    \part \textbf{(2 คะแนน)} จงแสดงวิธีการใช้\mbox{เมทริกซ์} $P$ และ $D$ นี้ในการคำนวณหา\mbox{เมทริกซ์} $A^5$
    \vspace{\stretch{1}}
\end{parts}

\question \textbf{(11 คะแนน)} กำหนดให้
\[ Q(x,y) = 7x^2 + 13y^2 + 8xy \]
อยู่ในรูปแบบกำลังสอง (in the quadratic form)
\begin{parts}
    \part \textbf{(3 คะแนน)} จงหา\mbox{เมทริกซ์}สมมาตร $A$ ที่สอดคล้องกับรูปแบบกำลังสองนี้
    \vspace{5cm}
    \part \textbf{(4 คะแนน)} จงหา\mbox{เมทริกซ์}เชิงตั้งฉาก $P$ (orthogonal matrix) ที่ทำให้ $P^T A P = D$ เป็น\mbox{เมทริกซ์}\mbox{ทแยงมุม}
    \vspace{\stretch{1}}
    \newpage
    \part \textbf{(4 คะแนน)} จงทำการเปลี่ยนตัวแปร $\vec{x} = P \vec{y}$ เพื่อให้รูปแบบกำลังสองของตัวแปรใหม่ ไม่มีเทอมผลคูณไขว้ปรากฏอยู่ (contains no cross-product terms)
    \vspace{\stretch{1}}
\end{parts}

% Question 5
\question \textbf{(7 คะแนน)} {\addfontfeature{LetterSpace=-3} การหมุนภาพในระนาบ (Rotation in 2D) ในระบบคอมพิวเตอร์กราฟิก คือการหมุนวัตถุรอบ\mbox{จุดกำเนิด}} (Rotation) ซึ่งถือเป็น Linear Transformation พื้นฐานที่สำคัญ หากกำหนดให้ $T : \mathbb{R}^2 \to \mathbb{R}^2$ เป็นฟังก์ชันการแปลงเชิงเส้นที่ทำการหมุน\mbox{เวกเตอร์} $\vec{x}$ ทวนเข็มนาฬิกาเป็นมุม $\theta$ (ในหน่วย เรเดียน) ไปยังจุดใหม่ $\vec{x}' = T(\vec{x})$ ความสัมพันธ์นี้สามารถเขียนในรูป\mbox{สมการ\mbox{เมทริกซ์}}ได้เป็น
\[ T(\vec{x}) = R_\theta \vec{x} \]
โดยที่ $R_\theta$ คือ Rotation Matrix:
\[ R_\theta = \begin{bmatrix} \operatorname{\text{\fontfamily{cmr}\selectfont cos}}\theta & -\operatorname{\text{\fontfamily{cmr}\selectfont sin}}\theta \\ \operatorname{\text{\fontfamily{cmr}\selectfont sin}}\theta & \operatorname{\text{\fontfamily{cmr}\selectfont cos}}\theta \end{bmatrix} \]
ซึ่ง\mbox{เมทริกซ์}นี้สร้างขึ้นจากการพิจารณาภาพของ\mbox{เวกเตอร์}ฐานหลัก $\vec{e}_1 = \begin{bmatrix} 1 \\ 0 \end{bmatrix}$ และ $\vec{e}_2 = \begin{bmatrix} 0 \\ 1 \end{bmatrix}$ ภายใต้การหมุน

\begin{parts}
    \part \textbf{(2 คะแนน)} จงหา\mbox{เวกเตอร์}ผลลัพธ์ $\vec{x}'$ จากการหมุน\mbox{เวกเตอร์} ${\vec{x} = \begin{bmatrix} 2 \\ 4 \end{bmatrix}}$ ทวนเข็มนาฬิกาเป็นมุม $\theta = \dfrac{\pi}{3}$
    \vspace{\stretch{1}}
    
    \part \textbf{(3 คะแนน)} จงหามุม $\theta$ ที่ทำให้การหมุนภาพเปลี่ยนพิกัดจาก ${\vec{x} = \begin{bmatrix} 1 \\ 1 \end{bmatrix}}$ ไปยัง ${\vec{x}' = \begin{bmatrix} -1 \\ 1 \end{bmatrix}}$
    \vspace{\stretch{1}}
    
    \part \textbf{(2 คะแนน)} จงพิสูจน์ว่า หากกำหนดฟังก์ชันการหมุนด้วยมุม $\theta = 2n\pi$ เมื่อ $n$ เป็นจำนวนเต็มใด ๆ \mbox{เมทริกซ์}การแปลง $R_\theta$ จะสมมูลกับ \mbox{เมทริกซ์}เอกลักษณ์ (Identity Matrix)
    \vspace{\stretch{1}}
\end{parts}
